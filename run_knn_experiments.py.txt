#!/usr/bin/env python3
"""Ejecuta el estudio de k-NN con validación cruzada anidada.

Para cada dataset CSV y cada fold externo evalúa cuatro configuraciones:
C1: sin normalización y sin balanceo
C2: con StandardScaler y sin balanceo
C3: sin normalización y con SMOTE
C4: con StandardScaler y con SMOTE

El mejor k se selecciona mediante CV interna usando macro-F1. Las métricas finales
se calculan exclusivamente en el fold externo de prueba.
"""

from __future__ import annotations

import argparse
import json
import logging
import math
import sys
from collections import Counter
from dataclasses import asdict, dataclass
from itertools import product
from pathlib import Path
from typing import Any, Iterable

import numpy as np
import pandas as pd
from imblearn.over_sampling import RandomOverSampler, SMOTE
from imblearn.pipeline import Pipeline
from sklearn.metrics import (
    accuracy_score,
    balanced_accuracy_score,
    f1_score,
    matthews_corrcoef,
    precision_score,
    recall_score,
)
from sklearn.model_selection import GridSearchCV, StratifiedKFold
from sklearn.neighbors import KNeighborsClassifier
from sklearn.preprocessing import LabelEncoder, StandardScaler


@dataclass(frozen=True)
class Settings:
    k_min: int = 1
    k_max: int = 10
    outer_folds: int = 5
    inner_folds: int = 5
    seed: int = 2026
    selection_metric: str = "f1_macro"
    n_jobs: int = -1
    adaptive_folds: bool = True
    smote_default_k_neighbors: int = 5
    fallback_random_oversampling: bool = True


CONFIGURATIONS = {
    "C1_raw_unbalanced": {"normalize": False, "balance": False},
    "C2_scaled_unbalanced": {"normalize": True, "balance": False},
    "C3_raw_smote": {"normalize": False, "balance": True},
    "C4_scaled_smote": {"normalize": True, "balance": True},
}


def read_csv_robust(path: Path) -> pd.DataFrame:
    df = pd.read_csv(path)
    if df.shape[1] == 1:
        df_alt = pd.read_csv(path, sep=None, engine="python")
        if df_alt.shape[1] > 1:
            df = df_alt
    return df


def class_counts(y: np.ndarray) -> Counter:
    return Counter(y.tolist())


def min_class_count(y: np.ndarray) -> int:
    counts = class_counts(y)
    return min(counts.values()) if counts else 0


def choose_n_splits(requested: int, y: np.ndarray, adaptive: bool) -> int:
    minimum = min_class_count(y)
    if minimum < 2:
        raise ValueError("Alguna clase tiene menos de dos instancias.")
    if requested <= minimum:
        return requested
    if not adaptive:
        raise ValueError(
            f"Se solicitaron {requested} folds, pero la clase menor solo tiene {minimum} casos."
        )
    return minimum


def validate_dataset(df: pd.DataFrame, path: Path) -> tuple[np.ndarray, np.ndarray, LabelEncoder]:
    if df.shape[1] < 2:
        raise ValueError("Se requieren al menos un predictor y una columna de clase.")
    if df.isna().any().any():
        raise ValueError("El dataset contiene valores ausentes.")

    X_df = df.iloc[:, :-1]
    y_raw = df.iloc[:, -1]

    non_numeric = [c for c in X_df.columns if not pd.api.types.is_numeric_dtype(X_df[c])]
    if non_numeric:
        raise ValueError(f"Se detectaron predictores no numéricos: {non_numeric}")

    X = X_df.to_numpy(dtype=float, copy=True)
    if not np.isfinite(X).all():
        raise ValueError("Los predictores contienen valores infinitos o no finitos.")

    encoder = LabelEncoder()
    y = encoder.fit_transform(y_raw.astype(str))
    if len(encoder.classes_) < 2:
        raise ValueError("La variable de clase debe contener al menos dos clases.")

    return X, y, encoder


def make_inner_splits(
    X_train: np.ndarray,
    y_train: np.ndarray,
    requested: int,
    seed: int,
    adaptive: bool,
) -> tuple[int, list[tuple[np.ndarray, np.ndarray]]]:
    n_splits = choose_n_splits(requested, y_train, adaptive)
    splitter = StratifiedKFold(n_splits=n_splits, shuffle=True, random_state=seed)
    splits = list(splitter.split(X_train, y_train))
    return n_splits, splits


def safe_smote_sampler(
    y_train: np.ndarray,
    inner_splits: Iterable[tuple[np.ndarray, np.ndarray]],
    seed: int,
    default_k: int,
    fallback: bool,
) -> tuple[Any, str, int | None]:
    """Elige un k_neighbors válido para todos los entrenamientos internos.

    Si alguna clase queda con una sola instancia dentro de un entrenamiento interno,
    SMOTE no puede interpolar. En ese caso se usa RandomOverSampler cuando el
    fallback está habilitado y se deja constancia en los resultados.
    """
    minimum = min(min_class_count(y_train[train_idx]) for train_idx, _ in inner_splits)
    smote_k = min(default_k, minimum - 1)

    if smote_k >= 1:
        return (
            SMOTE(random_state=seed, k_neighbors=smote_k, sampling_strategy="auto"),
            "SMOTE",
            smote_k,
        )

    if fallback:
        return (
            RandomOverSampler(random_state=seed, sampling_strategy="auto"),
            "RandomOverSampler_fallback",
            None,
        )

    raise ValueError(
        "SMOTE no es aplicable porque alguna clase tiene menos de dos casos "
        "en un entrenamiento interno."
    )


def build_pipeline(
    normalize: bool,
    balance: bool,
    sampler: Any | None,
) -> Pipeline:
    steps: list[tuple[str, Any]] = [
        ("scaler", StandardScaler() if normalize else "passthrough"),
        ("sampler", sampler if balance else "passthrough"),
        ("knn", KNeighborsClassifier()),
    ]
    return Pipeline(steps=steps)


def available_k_values(
    settings: Settings,
    inner_splits: Iterable[tuple[np.ndarray, np.ndarray]],
) -> list[int]:
    smallest_training_size = min(len(train_idx) for train_idx, _ in inner_splits)
    upper = min(settings.k_max, smallest_training_size)
    values = list(range(settings.k_min, upper + 1))
    if not values:
        raise ValueError(
            f"No hay valores válidos de k en [{settings.k_min}, {settings.k_max}]."
        )
    return values


def metric_values(y_true: np.ndarray, y_pred: np.ndarray) -> dict[str, float]:
    return {
        "accuracy": float(accuracy_score(y_true, y_pred)),
        "balanced_accuracy": float(balanced_accuracy_score(y_true, y_pred)),
        "macro_f1": float(f1_score(y_true, y_pred, average="macro", zero_division=0)),
        "weighted_f1": float(
            f1_score(y_true, y_pred, average="weighted", zero_division=0)
        ),
        "macro_precision": float(
            precision_score(y_true, y_pred, average="macro", zero_division=0)
        ),
        "macro_recall": float(
            recall_score(y_true, y_pred, average="macro", zero_division=0)
        ),
        "mcc": float(matthews_corrcoef(y_true, y_pred)),
    }


def append_rows(path: Path, rows: list[dict[str, Any]]) -> None:
    if not rows:
        return
    path.parent.mkdir(parents=True, exist_ok=True)
    frame = pd.DataFrame(rows)
    frame.to_csv(path, mode="a", header=not path.exists(), index=False)


def load_completed(path: Path) -> set[tuple[str, int, str]]:
    if not path.exists():
        return set()
    df = pd.read_csv(path, usecols=["dataset", "outer_fold", "configuration"])
    return {
        (str(r.dataset), int(r.outer_fold), str(r.configuration))
        for r in df.itertuples(index=False)
    }


def ensure_same_settings(output_dir: Path, settings: Settings) -> None:
    metadata_path = output_dir / "run_settings.json"
    current = asdict(settings)
    if metadata_path.exists():
        existing = json.loads(metadata_path.read_text(encoding="utf-8"))
        if existing != current:
            raise ValueError(
                "La carpeta de salida contiene resultados con otra configuración. "
                "Use una carpeta nueva o restablezca exactamente los mismos parámetros."
            )
    else:
        metadata_path.write_text(
            json.dumps(current, indent=2, ensure_ascii=False), encoding="utf-8"
        )


def dataset_summary_row(
    dataset: str,
    path: Path,
    X: np.ndarray,
    y: np.ndarray,
    encoder: LabelEncoder,
    requested_outer_folds: int,
    used_outer_folds: int,
) -> dict[str, Any]:
    counts = class_counts(y)
    smallest = min(counts.values())
    largest = max(counts.values())
    decoded_counts = {
        str(encoder.inverse_transform([label])[0]): int(count)
        for label, count in sorted(counts.items())
    }
    return {
        "dataset": dataset,
        "file": path.name,
        "n_instances": int(X.shape[0]),
        "n_features": int(X.shape[1]),
        "n_classes": int(len(counts)),
        "min_class_count": int(smallest),
        "max_class_count": int(largest),
        "imbalance_ratio": float(largest / smallest),
        "requested_outer_folds": requested_outer_folds,
        "used_outer_folds": used_outer_folds,
        "class_distribution": json.dumps(decoded_counts, ensure_ascii=False),
    }


def aggregate_results(output_dir: Path) -> None:
    fold_path = output_dir / "fold_results.csv"
    if not fold_path.exists():
        return

    df = pd.read_csv(fold_path)
    metric_cols = [
        "accuracy",
        "balanced_accuracy",
        "macro_f1",
        "weighted_f1",
        "macro_precision",
        "macro_recall",
        "mcc",
    ]

    grouped = df.groupby(["dataset", "configuration"], as_index=False)
    means = grouped[metric_cols].mean().rename(
        columns={metric: f"{metric}_mean" for metric in metric_cols}
    )
    stds = grouped[metric_cols].std(ddof=1).rename(
        columns={metric: f"{metric}_std" for metric in metric_cols}
    )

    def mode_smallest(series: pd.Series) -> int:
        counts = series.value_counts()
        max_count = counts.max()
        return int(min(counts[counts == max_count].index))

    extra = grouped.agg(
        n_outer_folds=("outer_fold", "nunique"),
        k_mode=("best_k", mode_smallest),
        k_mean=("best_k", "mean"),
        k_std=("best_k", "std"),
        best_inner_score_mean=("best_inner_score", "mean"),
    )

    result = means.merge(stds, on=["dataset", "configuration"]).merge(
        extra, on=["dataset", "configuration"]
    )
    result.to_csv(output_dir / "dataset_results.csv", index=False)


def run_dataset(
    path: Path,
    output_dir: Path,
    settings: Settings,
    completed: set[tuple[str, int, str]],
) -> None:
    dataset = path.stem
    logging.info("Procesando %s", dataset)

    df = read_csv_robust(path)
    X, y, encoder = validate_dataset(df, path)

    outer_n = choose_n_splits(settings.outer_folds, y, settings.adaptive_folds)
    outer_cv = StratifiedKFold(
        n_splits=outer_n, shuffle=True, random_state=settings.seed
    )
    outer_splits = list(outer_cv.split(X, y))

    summary_path = output_dir / "dataset_summary.csv"
    if not summary_path.exists() or dataset not in set(
        pd.read_csv(summary_path, usecols=["dataset"])["dataset"].astype(str)
    ):
        append_rows(
            summary_path,
            [
                dataset_summary_row(
                    dataset,
                    path,
                    X,
                    y,
                    encoder,
                    settings.outer_folds,
                    outer_n,
                )
            ],
        )

    fold_results_path = output_dir / "fold_results.csv"
    inner_results_path = output_dir / "inner_search_results.csv"

    for fold_index, (train_idx, test_idx) in enumerate(outer_splits, start=1):
        X_train, X_test = X[train_idx], X[test_idx]
        y_train, y_test = y[train_idx], y[test_idx]

        inner_n, inner_splits = make_inner_splits(
            X_train,
            y_train,
            settings.inner_folds,
            seed=settings.seed + fold_index,
            adaptive=settings.adaptive_folds,
        )
        k_values = available_k_values(settings, inner_splits)

        for config_name, flags in CONFIGURATIONS.items():
            key = (dataset, fold_index, config_name)
            if key in completed:
                logging.info("Omitido por reanudación: %s fold %s %s", dataset, fold_index, config_name)
                continue

            sampler = None
            sampler_name = "None"
            smote_k: int | None = None
            if flags["balance"]:
                sampler, sampler_name, smote_k = safe_smote_sampler(
                    y_train,
                    inner_splits,
                    seed=settings.seed + fold_index,
                    default_k=settings.smote_default_k_neighbors,
                    fallback=settings.fallback_random_oversampling,
                )

            pipeline = build_pipeline(
                normalize=flags["normalize"],
                balance=flags["balance"],
                sampler=sampler,
            )
            search = GridSearchCV(
                estimator=pipeline,
                param_grid={"knn__n_neighbors": k_values},
                scoring=settings.selection_metric,
                cv=inner_splits,
                refit=True,
                n_jobs=settings.n_jobs,
                return_train_score=False,
                error_score="raise",
            )
            search.fit(X_train, y_train)
            y_pred = search.predict(X_test)
            metrics = metric_values(y_test, y_pred)

            best_k = int(search.best_params_["knn__n_neighbors"])
            fold_row: dict[str, Any] = {
                "dataset": dataset,
                "configuration": config_name,
                "normalize": bool(flags["normalize"]),
                "balance": bool(flags["balance"]),
                "sampler_used": sampler_name,
                "smote_k_neighbors": smote_k,
                "outer_fold": fold_index,
                "outer_folds_used": outer_n,
                "inner_folds_used": inner_n,
                "train_size": int(len(train_idx)),
                "test_size": int(len(test_idx)),
                "best_k": best_k,
                "best_inner_score": float(search.best_score_),
                "selection_metric": settings.selection_metric,
                **metrics,
            }
            append_rows(fold_results_path, [fold_row])

            inner_rows: list[dict[str, Any]] = []
            cv_results = search.cv_results_
            for i, params in enumerate(cv_results["params"]):
                inner_rows.append(
                    {
                        "dataset": dataset,
                        "configuration": config_name,
                        "outer_fold": fold_index,
                        "k": int(params["knn__n_neighbors"]),
                        "mean_inner_score": float(cv_results["mean_test_score"][i]),
                        "std_inner_score": float(cv_results["std_test_score"][i]),
                        "rank_inner_score": int(cv_results["rank_test_score"][i]),
                        "selection_metric": settings.selection_metric,
                    }
                )
            append_rows(inner_results_path, inner_rows)
            completed.add(key)

            logging.info(
                "%s | fold %d | %s | k=%d | macro-F1=%.4f",
                dataset,
                fold_index,
                config_name,
                best_k,
                metrics["macro_f1"],
            )


def main() -> None:
    parser = argparse.ArgumentParser(
        description="Estudio de k-NN con validación cruzada anidada."
    )
    parser.add_argument("input_dir", type=Path, help="Carpeta con los CSV.")
    parser.add_argument("output_dir", type=Path, help="Carpeta de resultados.")
    parser.add_argument("--k-min", type=int, default=1)
    parser.add_argument("--k-max", type=int, default=10)
    parser.add_argument("--outer-folds", type=int, default=5)
    parser.add_argument("--inner-folds", type=int, default=5)
    parser.add_argument("--seed", type=int, default=2026)
    parser.add_argument("--n-jobs", type=int, default=-1)
    parser.add_argument(
        "--strict-folds",
        action="store_true",
        help="No adaptar el número de folds cuando una clase tiene pocos casos.",
    )
    parser.add_argument(
        "--no-smote-fallback",
        action="store_true",
        help="No usar RandomOverSampler cuando SMOTE sea matemáticamente imposible.",
    )
    args = parser.parse_args()

    if args.k_min < 1 or args.k_max < args.k_min:
        raise SystemExit("El rango de k no es válido.")
    if args.outer_folds < 2 or args.inner_folds < 2:
        raise SystemExit("La validación cruzada requiere al menos dos folds.")

    settings = Settings(
        k_min=args.k_min,
        k_max=args.k_max,
        outer_folds=args.outer_folds,
        inner_folds=args.inner_folds,
        seed=args.seed,
        n_jobs=args.n_jobs,
        adaptive_folds=not args.strict_folds,
        fallback_random_oversampling=not args.no_smote_fallback,
    )

    args.output_dir.mkdir(parents=True, exist_ok=True)
    ensure_same_settings(args.output_dir, settings)

    logging.basicConfig(
        level=logging.INFO,
        format="%(asctime)s | %(levelname)s | %(message)s",
        handlers=[
            logging.FileHandler(args.output_dir / "experiment.log", encoding="utf-8"),
            logging.StreamHandler(sys.stdout),
        ],
    )

    files = sorted(args.input_dir.glob("*.csv"))
    if not files:
        raise SystemExit(f"No se encontraron CSV en: {args.input_dir}")

    completed = load_completed(args.output_dir / "fold_results.csv")
    errors: list[dict[str, Any]] = []

    for path in files:
        try:
            run_dataset(path, args.output_dir, settings, completed)
        except Exception as exc:  # noqa: BLE001
            logging.exception("Error en %s", path.name)
            errors.append(
                {
                    "dataset": path.stem,
                    "file": path.name,
                    "error_type": type(exc).__name__,
                    "message": str(exc),
                }
            )
            append_rows(args.output_dir / "errors.csv", [errors[-1]])

    aggregate_results(args.output_dir)
    logging.info("Proceso terminado. Resultados en %s", args.output_dir.resolve())


if __name__ == "__main__":
    main()
